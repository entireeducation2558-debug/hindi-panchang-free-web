विक्रम संवत (Vikram Samvat)

सम्राट विक्रमादित्य द्वारा ५७ ईसा पूर्व आरम्भ किया गया यह संवत ग्रेगोरियन कैलेंडर से ५७ वर्ष आगे चलता है। इसका नववर्ष चैत्र शुक्ल प्रतिपदा (गुड़ी पड़वा) से आरम्भ होता है।

पंचांग के पाँच अंगों का महत्व

'पंचांग' का शाब्दिक अर्थ है पाँच अंग—तिथि, वार, नक्षत्र, योग और करण। किसी भी मांगलिक कार्य जैसे विवाह, गृह प्रवेश, मुंडन या व्यापार आरम्भ करने से पहले इन पाँचों की शुद्धि देखी जाती है।

राहुकाल और अभिजीत मुहूर्त क्या हैं?

प्रत्येक दिन सूर्योदय और सूर्यास्त के बीच लगभग ९० मिनट का समय राहुकाल कहलाता है, जिसमें नए शुभ कार्य वर्जित हैं। वहीं दोपहर के समय लगभग ४८ मिनट का अभिजीत मुहूर्त सभी दोषों को नष्ट करने वाला माना जाता है।

# app.py - Production Hindi Panchang Flask API with Swiss Ephemeris (pyswisseph)
from flask import Flask, request, jsonify, Response
from datetime import datetime, timedelta
import swisseph as swe
import sqlite3

app = Flask(__name__)
swe.set_sid_mode(swe.SIDM_LAHIRI)  # Lahiri Ayanamsa for Vedic Panchang

TITHIS_HI = [
    "प्रतिपदा", "द्वितीया", "तृतीया", "चतुर्थी", "पंचमी", "षष्ठी", "सप्तमी", "अष्टमी",
    "नवमी", "दशमी", "एकादशी", "द्वादशी", "त्रयोदशी", "चतुर्दशी", "पूर्णिमा",
    "प्रतिपदा", "द्वितीया", "तृतीया", "चतुर्थी", "पंचमी", "षष्ठी", "सप्तमी", "अष्टमी",
    "नवमी", "दशमी", "एकादशी", "द्वादशी", "त्रयोदशी", "चतुर्दशी", "अमावस्या"
]

NAKSHATRAS_HI = [
    "अश्विनी", "भरणी", "कृत्तिका", "रोहिणी", "मृगशिरा", "आर्द्रा", "पुनर्वसु", "पुष्य", "आश्लेषा",
    "मघा", "पूर्वाफाल्गुनी", "उत्तराफाल्गुनी", "हस्त", "चित्रा", "स्वाती", "विशाखा", "अनुराधा", "ज्येष्ठा",
    "मूल", "पूर्वाषाढ़ा", "उत्तराषाढ़ा", "श्रवण", "धनिष्ठा", "शतभिषा", "पूर्वभाद्रपद", "उत्तरभाद्रपद", "रेवती"
]

@app.route('/api/panchang', methods=['GET'])
def get_panchang():
    date_str = request.args.get('date', datetime.now().strftime('%Y-%m-%d'))
    lat = float(request.args.get('lat', 27.1767))
    lon = float(request.args.get('lon', 78.0081))

    dt = datetime.strptime(date_str, '%Y-%m-%d')
    jd = swe.julday(dt.year, dt.month, dt.day, 0.5)  # 06:00 AM IST

    sun_lon = swe.calc_ut(jd, swe.SUN, swe.FLG_SIDEREAL)[0][0]
    moon_lon = swe.calc_ut(jd, swe.MOON, swe.FLG_SIDEREAL)[0][0]

    diff = (moon_lon - sun_lon) % 360
    tithi_idx = int(diff // 12)
    paksha = "शुक्ल पक्ष" if tithi_idx < 15 else "कृष्ण पक्ष"
    nakshatra_idx = int(moon_lon // (360 / 27))

    return jsonify({
        "date": date_str,
        "paksha": paksha,
        "tithi": TITHIS_HI[tithi_idx],
        "nakshatra": NAKSHATRAS_HI[nakshatra_idx],
        "ayanamsa": "Lahiri"
    })

@app.route('/robots.txt')
def robots():
    return Response("User-agent: *\nAllow: /\nSitemap: https://hindipanchang.example.com/sitemap.xml", mimetype='text/plain')

@app.route('/sitemap.xml')
def sitemap():
    xml = '''

  https://hindipanchang.example.com/daily1.0
'''
    return Response(xml, mimetype='application/xml')

if __name__ == '__main__':
    app.run(debug=True, port=5000)
